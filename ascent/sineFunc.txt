

anim_curve_sin_values_start .equ vram ;140 bytes
;sine phase 0 speed half, amp 127-1, then increments phases, then speeds

anim_curve_lin_values_start .equ anim_curve_sin_values_start + 140; 140bytes


update_sine_vals:
	ld bc,0
	
	ld a, (frameCount+1);middle byte 
	srl a ;put bit 0 to carry flag 
	ld a, (frameCount)
	rra ;a holds frame counter/2, ~16 second timer
	ld de,anim_curve_sin_values_start
	exx 
	ld de, anim_curve_lin_values_start
	exx 
	ex af,af' ;' ;in
	ld a,20
	
update_sine_loop:
	ex af,af' ;' ;out	
	
	ld c,a
	call sine_apply_amplitudes
	ld a,c ;restore original value
	;A holds frame count for linear curve
	
	ld c,a ;save value
	exx 
	call lin_apply_amplitudes
	exx 
	ld a,c 
	
	ld c,64 ;add quarter phase offset 
	add a,c 
	ex af,af' ;' ;in
	dec a 
	ret z 
	ld c,a
	and %00000011 ;check last 2 bits
	ld a,c
	jp nz,update_sine_loop
	;otherwise, a is multiple of 4, multiply clock speed 
	ex af, af' ;' ;out
	sla a ;mult by 2 
	ex af, af' ;' ;in
	jp update_sine_loop
	
	
	
sine_apply_amplitudes: ;a and c holds scaled frame count,de holds ptr to anim curve value destination
	ld hl, sine
	add hl,bc 
	ld a,(hl); a holds sine value 
lin_apply_amplitudes:
	srl a ;div 2 
linrep_apply_amplitudes:
	ld (de),a ;amplitude of 127 
	inc de
	srl a
	ld (de),a ;amplitude of 63 
	inc de 
	srl a ;amplitude of 31 
	ld (de),a 
	inc de 
	srl a ;amplitude of 15 
	ld (de),a 
	inc de
	srl a ;amplitude of 7 
	ld (de),a 
	inc de
	srl a ;amplitude of 3 
	ld (de),a 
	inc de
	srl a ;amplitude of 1 
	ld (de),a
	inc de
	ret
