
;Scrolling BG thoughts
;Half vertical resolution of BG layer 
;Rows in BG buffer are twice as long 
;When copying from BG buffer to draw buffer, offset where you are copying from for each row
;Each row will be copied individaully, and doubled 

;When drawing lines to BG buffers, X pos of sprites should be doubled so full width (640px) can be achived with 1 byte pos data 

;When transitioning modes, BG_draw_buffer should be set back to top of buffer

draw_bg_scroll:; function for drawing background buffer to the draw buffer, clearing the frame for fg sprites to be drawn over 
	
	;ld a, $FF 
	;call write_a_to_ram
	
	;ld hl,(BG_draw_buffer)
	;call write_hl_to_ram
	
	ld a,120
	ld (draw_bg_scroll_num_loops),a 
	ld hl, (draw_buffer)
	push hl 
bg_scroll_buffer_x_offset .equ $ + 1
	ld de,319
	ld hl, 320 ;max x offset 
	scf 
	ccf 
	sbc hl,de ;hl holds length of first ldir 
	;need to limit max to 160 
	ld a,h 
	and a 
	jp nz, draw_bg_scroll_lim
	ld a,l 
	cp 160
	jp nc,draw_bg_scroll_lim
	jp draw_bg_scroll_nolim
draw_bg_scroll_lim:
	ld hl,160
draw_bg_scroll_nolim:
	ld (draw_bg_scroll_loop_ldir1_len), hl
	
	;call write_hl_to_ram
	
	ld hl,(BG_draw_buffer)
	add hl,de
	ld (draw_bg_scroll_BG_buffer_line),hl 
	
	;call write_hl_to_ram
	
	exx ;in
	;line duplicating in shadow registers 
	ld hl, (draw_buffer) 
	ld bc, 160 
	add hl,bc 
	ex de,hl 
	ld hl, (draw_buffer)
	;hl' holds line 1 in draw buffer to copy from, de' holds next line down to copy to 
	
draw_bg_scroll_loop:
	exx ;out 
	;need to track line in BG buffer, and last position in draw buffer 
	;maybe use shadow registers to do line duplicating part? 

	;ld hl, (BG_draw_buffer) ;hl holds uppermost line of BG in buffer

draw_bg_scroll_loop_ldir1_len .equ $ + 1
	ld bc, 0
	;bc holds length of first ldir
draw_bg_scroll_BG_buffer_line .equ $ + 1	
	ld hl,0 ;start of first ldir in bg buffer 
	pop de ;draw buffer pos 
	ld a,c ;save length for later 
	ldir 
	ld c,a ;bc now holds first ldir length  
	ld a, 160
	sub c
	jp z,draw_bg_scroll_second_ldir_skip
	;hl should be at beginning of next row in BG buffer after doing ldir 
	;can subtract 320 to get to start of current row 
	ld bc,320
	scf 
	ccf 
	sbc hl,bc
	ld b,0
	ld c,a 
	;bc holds length of second ldir
	
	ldir 
	
draw_bg_scroll_second_ldir_skip:
	;Move down 1 row in BG buffer 
	;check if need to wrap to top of buffer 
	ld hl,(draw_bg_scroll_BG_buffer_line)
	ld bc, $D52AC0 ;if HL exceeds this value, should loop to top of buffer 
	sbc hl,bc 
	jp c, draw_bg_scroll_nowrap
	ld hl, $D49600 ;start of BG buffer
bg_scroll_buffer_x_offset_2 .equ $ + 1
	ld bc,319
	add hl,bc
	jp draw_bg_scroll_yeswrap
draw_bg_scroll_nowrap:
	ld hl,(draw_bg_scroll_BG_buffer_line)
	ld bc, 320 
	add hl, bc 
draw_bg_scroll_yeswrap:	
	ld (draw_bg_scroll_BG_buffer_line), hl 

	exx ; in 
	;duplicate row 
	ld bc, 160 
	ldir 
	;need to add 160 to hl and de to setup for next row 
	ld bc, 160 
	add hl,bc 
	ex de, hl 
	add hl,bc 
	ex de,hl 
	;hl holds new row of draw buffer to write to 
	push hl

draw_bg_scroll_num_loops .equ $ + 1
	ld a,0
	dec a 
	ld (draw_bg_scroll_num_loops),a 
	jp nz, draw_bg_scroll_loop 
	
	pop hl ;need to balance stack
	
	ld hl,(bg_scroll_buffer_x_offset)
	dec hl
	ld (bg_scroll_buffer_x_offset), hl 
	ld (bg_scroll_buffer_x_offset_2), hl 
	ld bc,1
	sbc hl,bc
	ret nc
	ld hl,319 
	ld (bg_scroll_buffer_x_offset), hl 
	ld (bg_scroll_buffer_x_offset_2), hl
	ret
	
	
	
set_bg_scroll_mode:

	;only called once when transition occurs
	;should also reset bg_cam_pos 
	ld hl,(cam_pos)
	srl h
	rr l
	srl h
	rr l
	srl h
	rr l ;div by 8
	;hl holds bg cam pos 
	ld (bg_cam_pos),hl
	ld hl,BG_buffer ;start of bg drawing buffer 
	ld (BG_draw_buffer),hl ;set current bg line drawing location 
	
	
	ld a, %11000000 ;ret nz 
	ld (bg_scroll_mod_1),a 
	ld hl, 320 
	ld (bg_scroll_mod_3),hl
	ld (bg_scroll_mod_4),hl
	;ld (bg_scroll_mod_8),hl
	ld (bg_scroll_mod_9),hl
	ld hl, $D52AC0
	ld (bg_scroll_mod_5),hl 
	ld a, 120 
	ld (bg_scroll_mod_6),a 
	ld a, 119
	ld (bg_scroll_mod_7),a 
	ld a, %00011001 ;add hl,de 
	ld (bg_scroll_mod_10),a 
	ld (bg_scroll_mod_11),a 
	ld a,%11011010 ;jp c
	ld (bg_scroll_mod_12),a 
	ret	
	
reset_bg_scroll_mode:
	
	ld hl,(cam_pos)
	srl h
	rr l
	srl h
	rr l
	srl h
	rr l ;div by 8
	;hl holds bg cam pos 
	ld (bg_cam_pos),hl
	
	ld hl,BG_buffer ;start of bg drawing buffer 
	ld (BG_draw_buffer),hl ;set current bg line drawing location 

	ld a, %00000000 ;nop
	ld (bg_scroll_mod_1),a 
	ld (bg_scroll_mod_10),a 
	ld (bg_scroll_mod_11),a 
	ld hl, 160 
	ld (bg_scroll_mod_3),hl
	ld (bg_scroll_mod_4),hl
	;ld (bg_scroll_mod_8),hl
	ld (bg_scroll_mod_9),hl
	ld hl, $d52b60
	ld (bg_scroll_mod_5),hl 
	ld a, 240
	ld (bg_scroll_mod_6),a 
	ld a,239
	ld (bg_scroll_mod_7),a 
	ld a,%11010010 ;jp nc
	ld (bg_scroll_mod_12),a 
	ret
