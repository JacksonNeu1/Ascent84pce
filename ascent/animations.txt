 
update_animations:
	;Will update all animations every frame. Not significant performance impact
	
	;POSITION
	;load hl with data pointer (dataframe or sprite group)
	;ld bc with number of bytes to target property
	;add hl bc 

	;ld a with desired sine freq,phase,and amplitude 
	;add a,xx ;Base value
	;ld (hl),a

	;7 bytes per animated value
	ex af, af' ;'
	ld hl,position_animation_data
	ld a,(hl) ;replace w shadow bc for >256 vals 
	inc hl
update_animation_position_loop:
	ex af, af' ;'
	ld de,(hl);load address of property
	inc hl
	inc hl 
	inc hl 
	ld bc,(hl) ;address of anim curve 
	inc hl 
	inc hl 
	inc hl
	ld a,(bc) ;value of anim curve
	ld c, (hl) ;base value
	inc hl
	add a,c
	ld (de),a 
	ex af, af' ;'
	dec a 
	jp nz,update_animation_position_loop
	

; Enable sprites, 8 bytes per animated property 	
	ex af, af' ;'
	ld hl,disable_animation_data
	ld a,(hl)
	inc hl
update_animation_enable_loop:
	ex af, af' ;'
	ld de,(hl);load address of property
	inc hl
	inc hl 
	inc hl 
	ld bc,(hl) ;address of anim curve 
	inc hl 
	inc hl 
	inc hl
	ld a,(bc) ;value of anim curve
	ld c,a 
	ld a,(hl) ;bitmask (first bit = negate)
	inc hl 
	bit 7,a 
	jp nz,update_animation_enable_inverted
	and c 
	jp nz,update_animation_enable_height
	inc hl
	ex af, af' ;'
	dec a 
	jp nz,update_animation_enable_loop
	jp update_animation_sprite
	
update_animation_enable_inverted:
	and c 
	jp z,update_animation_enable_height
	
	inc hl
	ex af, af' ;'
	dec a 
	jp nz,update_animation_enable_loop
	jp update_animation_sprite
	
update_animation_enable_height:
	ld a,(hl)
	inc hl
	ld (de),a 
	ex af, af' ;'
	dec a 
	jp nz,update_animation_enable_loop

	;call write_a_to_ram
	
	
update_animation_sprite:	
	
	
	ex af, af' ;' ;in
	ld hl,sprite_animation_data
	ld a,(hl)
	inc hl
update_animation_sprite_loop:
	ex af, af' ;' ;out
	ld de,0 ;clear upper byte 
	ld bc,(hl) ;animation curve address
	inc hl 
	inc hl
	inc hl
	ld a,(bc) ;anim curve value (btwn 0 and 7)
	rlca ;double it and give it to the next person
	ld e,a 
	add hl,de ;jump to correct place
	neg ;a holds -curve value
	add a,15; a holds amt to jump to get past sprite table
	ld e,a 
	ld b,(hl);correct sprite index 
	inc hl 
	ld c,(hl) ;sprite height value 
	add hl,de ;to address of target property
	ld de,(hl)
	ld a,b
	ld (de),a ;sprite index 
	dec de
	dec de ;to height 
	ld a,c
	ld (de),a ;sprite height
	inc hl 
	inc hl 
	inc hl ;to next
	ex af, af' ;' ;in
	dec a 
	jp nz,update_animation_sprite_loop
	
;SPrite animations isnt updating height values in object data
;this mainly impacts when sprite is cutoff at top as height is used to calculate how many lines to draw
;otherwise for fullsprite all lines are drawn, height is irrelevant
;Easiest fix would be to have sprite anim include height data
;This would increase memory use of sprite animations 
;Also would need other way to do disable anims, as height would also be set by sprite anim 
;X and Y properties are also animated
;Actually should be OK to use height = 0 because disable anim is executed after sprite anims 
;But what about restoring default height value? would mess up value set by sprite anim 
;Use two loops? one for 'un disable' prior to sprite anims, then disable after sprite anims 
	
; Disable sprites, 8 bytes per animated property 	
	ex af, af' ;'
	ld hl,disable_animation_data
	ld a,(hl)
	inc hl
update_animation_disable_loop:
	ex af, af' ;'
	ld de,(hl);load address of property
	inc hl
	inc hl 
	inc hl 
	ld bc,(hl) ;address of anim curve 
	inc hl 
	inc hl 
	inc hl
	ld a,(bc) ;value of anim curve
	ld c,a 
	ld a,(hl)
	inc hl
	bit 7,a 
	jp nz,update_animation_disable_inverted
	and c  
	jp z,update_animation_disable_zero_height
	inc hl
	ex af, af' ;'
	dec a 
	jp nz,update_animation_disable_loop
	jp update_palette_anim
	
update_animation_disable_inverted:
	and c  
	jp nz,update_animation_disable_zero_height
	inc hl
	ex af, af' ;'
	dec a 
	jp nz,update_animation_disable_loop
	jp update_palette_anim
	
update_animation_disable_zero_height:
	ld a,0
	inc hl
	ld (de),a 
	ex af, af' ;'
	dec a 
	jp nz,update_animation_disable_loop

	;call write_a_to_ram
	
update_palette_anim:
	ld a, (cam_pos+1) ;cam frame #
	;call write_a_to_ram
	sub 134
	ret c ;return if frame is not >=136
	
	ld de, space_gradients
	;ld a, $00
	;call write_a_to_ram
	ld a, (lin_2_31_1) ;~4s timer
	ld hl,0
	ld l,a 
	ld h,20 
	mlt hl
	add hl,de ;hl points to correct part of gradient data

	;reorder gradient data such that ldir can be used for most of it
	
	;ldir 0,2,4,6

	ld de, temp_palette_buffer
	ld bc,8
	ldir 
	ld de, temp_palette_buffer+14
	ldi

	ld de, temp_palette_buffer+20
	ld bc,10

	ret
