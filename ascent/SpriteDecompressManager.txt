

decompress_queue_start .equ vram

decompress_queue_end .equ vram + (11*64) ;Max queue length of 64 (size of elements is always 11)
;D402C0
decompress_tail_ptr:
	.dl decompress_queue_start + 6  ;points to sprite being currently decompressed (parameters byte)

decompress_head_ptr: ;points to where next element will be added to queue
	.dl decompress_queue_start

;Decompress queue format
; .dl Sprite data ptr, decompress location, parameters (7 = fast, 6 = offset, 5 = flip, 432 = num of alt colors, 1 = started decompressing, 0 = finished decomp)

setup_decompress_queue: ;clears memory in queue
	ld a,$FF
	ld bc, 11*64
	ld hl, decompress_queue_start
	ld (decompress_head_ptr),hl ;Is this needed?
	call _MemSet
	ld hl, decompress_queue_start + 6 ;Is this needed?
	ld (decompress_tail_ptr),hl  ;Is this needed?
	ld a,0 
	ld (last_decompressed_cam_pos),a ;Is this needed?
	ret 


	
	


;Runs after frame has been rendered. Checks through decompress queue and continues sprite decompressions
	;if has ended, return. at end of queue
	;if not started, begin decompression and set start flag
	;if has started, but not ended, continue decompression 
	;if started and ended, move to next element 
continue_decompressions:
	ld hl, (decompress_tail_ptr)
	ld a,(hl) ;decompress params 
	bit 0,a 
	ret nz ;sprite is done decompressing, at end of queue 
	
	;testing
	push af
	ld a,$55
	call write_a_to_ram
	pop af
	
	
	
	
	bit 1,a
	jp z, cd_begin_decompression
	
	;has started,not ended 
	call sdcomp_continue 
	or a 
	ret nz ;sprite not yet done decompressing 

cd_done_decompressing:
	;sprite finished decompressing
	ld hl, (decompress_tail_ptr)	
	set 0,(hl) ;set finished flag
	ld a,(hl)
	ld de,11 

	add hl,de ;points to next element in queue
	
	push hl 
	ld de, decompress_queue_end + 6 
	scf
	ccf
	sbc hl,de ;check if need to wrap to start of queue
	pop hl
	jp nc, cd_back_to_start
	ld (decompress_tail_ptr),hl
	jp continue_decompressions ;Move on to next sprite 
	
cd_back_to_start:

	ld a, $5B
	call write_a_to_ram

	ld hl,decompress_queue_start + 6
	;ld de,6 ;tail points to the parameters byte
	;add hl,de
	ld (decompress_tail_ptr),hl
	jp continue_decompressions ;Move on to next sprite 

cd_begin_decompression:

	;testing
	;push af 
	;ld a,$AA
	;call write_a_to_ram
	;pop af
	
	;setup and begin decompressing
	set 1,(hl) ;set start flag
	ld c,a ;save byte 
	
	bit 7,c 
	call nz, sdcomp_set_fast
	bit 7,c 
	call z, sdcomp_reset_fast
	bit 6,c 
	call nz, sdcomp_set_offset
	bit 6,c
	call z, sdcomp_reset_offset
	bit 5,c 
	call nz, sdcomp_set_flip
	bit 5,c 
	call z, sdcomp_reset_flip
	
	call sdcomp_reset_alt_color
	inc hl ;start of alt color data 
	ld a,c 
	and %00011100
	call nz, sdcomp_set_alt_color
	
	
	ld hl, (decompress_tail_ptr) ;return to flags byte
	
	dec hl ;traverse back to decomp location
	dec hl 
	dec hl 
	ld de,(hl) ;load decompress location

	dec hl 
	dec hl 
	dec hl 
	ld hl,(hl) ;load sprite data ptr

	call sprite_decompress
	or a 
	ret nz ;sprite not yet done decompressing 
	jp cd_done_decompressing


;Checks to see if cam has moved to a new frame and decompressions need to be added to queue
check_for_decompress_calls:
	ld a, (last_decompressed_cam_pos)
	ld b,a ;put last decompressed pos in b
	ld a, (cam_pos + 1) ;Current frame cam is in
	ld (last_decompressed_cam_pos),a ;store this frame as the last decompressed frame 
	;need to check if cam has moved up or down since last time
	cp a,b 
	
	
	ret z ;if same, no new things to add to queue 
	jp nc, cfdc_cam_move_up ;camera has moved up since last time 
	
	;cam has moved down since last time 
	ld hl, decompress_calls_table_down
	jp cfdc_add_calls_to_queue
	
	
cfdc_cam_move_up:
	ld hl, decompress_calls_table_up

	;a = cam frame, hl = decompress call table (up or down)
cfdc_add_calls_to_queue:

	;call write_a_to_ram

	ld bc,0
	ld c,a ;current cam frame
	add hl,bc 
	add hl,bc 
	add hl,bc 
	ld hl,(hl) ;hl now points to correct decompress frame 
cfdc_direct_add_decompress_frame:

	ex af,af' ;'
	ld a,(hl) ;a' holds num elements 
	
	;call write_a_to_ram
	
	;call prgmpause
	
	inc a ;to be decd later
	ex af,af' ;'
	
	ld de, (decompress_head_ptr) ;position in decomp queue to add to 
	inc hl ;points to first sprite data 
cfdc_add_to_queue_loop:
	ex af,af' ;'
	;call write_a_to_ram
	dec a ;account for cases where frame has 0 elements 
	ret z
	ex af,af' ;'
	
	ldi ;copy compressed sprite name to decomp queue
	ldi 
	ldi 
	
	ld a,(hl) ;index in table 
	ld (cfdc_add_idx_in_tbl),a 
	inc hl 
	ld a,(hl) ;ld params ;params (7=isFast,6=isOffset,5=isFlip, 432 = num of alt colors, 10 = sprite tbl index )
	
	push hl
	ld bc,0 
	and %00000011 ;get table index
	ld c,a
	;c should hold sprite table index 
	ld hl,Sprite_Table_Table
	add hl,bc
	add hl,bc 
	add hl,bc ;inc to correct table
	ld hl,(hl)
cfdc_add_idx_in_tbl .equ $ + 1
	ld c,0
	add hl,bc
	add hl,bc 
	add hl,bc ;inc to decompress pos 
	ldi ;copy decompres pos to queue  
	ldi
	ldi
	pop hl 
	
	ld a,(hl) ;params byte 
	and %11111100 ;clear tbl index bits to be used for started/ended flags
	ld (de), a
	inc de ;start of next element (or color data)
	inc hl 
	and %00011100
	jp z,a2dq_0_alt_col
	bit 2,a 
	jp nz,a2dq_1_alt_col
	bit 3,a 
	jp nz,a2dq_2_alt_col

;4 alt colors	
	ldi 
	ldi
	ldi 
	ldi
	jp a2dq_post_alt_col
a2dq_2_alt_col:
	ldi
	ldi
	inc de 
	inc de
	jp a2dq_post_alt_col
a2dq_1_alt_col:
	ldi
	inc de 
	inc de 
	inc de
	jp a2dq_post_alt_col
a2dq_0_alt_col:	
	
	inc de 
	inc de 
	inc de 
	inc de
a2dq_post_alt_col:
	
	;Check if reached end of decomp queue
	push de 
	ex de,hl
	ld bc, decompress_queue_end
	scf
	ccf
	sbc hl,bc ;check if we have reached end of queue to wrap around
	ex de,hl
	pop de
	jp nc, a2dq_back_to_start
	
	ld (decompress_head_ptr),de
	;ld a,$AB ;testing
	;call write_a_to_ram
	jp cfdc_add_to_queue_loop
	
a2dq_back_to_start:

	ld a, $5C
	call write_a_to_ram

	ld de,decompress_queue_start
	ld (decompress_head_ptr),de
	jp cfdc_add_to_queue_loop	
	


last_decompressed_cam_pos:
	.db 0 ;camera frame last time checking for decomp was called
